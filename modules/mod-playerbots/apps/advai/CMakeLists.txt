cmake_minimum_required(VERSION 3.20)
project(playerbot_advai_tests LANGUAGES CXX)
enable_testing()
add_executable(advai_tests ../../src/Advai/AdvaiModel.cpp ../../src/Advai/AdvaiMetrics.cpp
               ../../src/Advai/AdvaiCognition.cpp ../../src/Advai/AdvaiDialogue.cpp AdvaiTests.cpp)
target_include_directories(advai_tests PRIVATE ../../src/Advai)
target_compile_features(advai_tests PRIVATE cxx_std_20)
target_compile_definitions(advai_tests PRIVATE ADVAI_ENGLISH_CATALOG="${CMAKE_CURRENT_SOURCE_DIR}/catalog/enUS.txt")
if(MSVC)
  target_compile_options(advai_tests PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(advai_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME advai_rules_planner_scheduler_lifecycle COMMAND advai_tests)
